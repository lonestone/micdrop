<ArchitectureSection

  intro="Micdrop orchestrates a complete voice conversation pipeline. Watch how voice data flows through each component in real-time, from your microphone to AI responses and back to your speakers."
>
  <ArchitectureStep title="🎤 Voice Input">

  </ArchitectureStep>
  <ArchitectureStep title="🤖 AI Processing">

  </ArchitectureStep>
  <ArchitectureStep title="🔊 Voice Output">
